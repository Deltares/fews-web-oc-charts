Move over any chart to inspect a line, stacked bars, or stacked areas.

Public API

const interaction = crosshair({ mode: 'xy' })
const plot = createPlot(host, { scales: { t, q }, axes, marks, plugins: [interaction] })
interaction.update({ mode: 'x' })
interaction.update({ snap: { mark: observations, axis: 'xy' } })

const barsTooltip = tooltip({ marks: [barMark], crosshair: barsGuide })
