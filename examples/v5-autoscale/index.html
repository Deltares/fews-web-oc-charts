      Explicit domains take precedence over every policy. Uncheck the override to resume the
      selected policy. Disable the time window to let time autoscale too. Visible extents use
      samples inside inclusive window boundaries, without interpolating between samples.
    

    
Public API

    
const t = time()
const q = linear({ autoscale: 'visible', defaultDomain: [-20, 120] })
// Bind scales and series through createPlot.
t.domain.set([windowStart, windowEnd])
q.update({ autoscale: 'data' })
q.domain.set([-25, 50]) // explicit override
q.update({ domain: undefined }) // resume the selected policy
t.update({ domain: undefined }) // resume time autoscaling
q.update({
  autoscale: ({ data }) => {
    if (!data) return undefined // use defaultDomain
    const limit = Math.max(...data.map((value) => Math.abs(Number(value))))
    return [-limit, limit]
  },
})
